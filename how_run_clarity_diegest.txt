# Startet clarity_digest.py für ein Fachgebiet (Ordner unter specialties/).
#
# Nutzung:  ./run_digest.sh <fachgebiet> [modus] [weitere Optionen für clarity_digest.py]
#
# Modi:
#   check   Parameterdateien prüfen, Journalnamen bei PubMed testen (kostenlos, kein State)
#   test    kleiner Lauf in specialties/<fach>/test_state, State bleibt unverändert (Standard)
#   prod    produktiver Wochenlauf: PDF + Word, State wird fortgeschrieben
#   rebuild Dokumente aus dem letzten Lauf neu bauen (keine PubMed-Abfrage, keine Bewertung)
#
# Beispiele:
#   ./run_digest.sh cardiology check
#   ./run_digest.sh cardiology test
#   ./run_digest.sh cardiology test --provider gemini
#   ./run_digest.sh cardiology prod --lang de --only "Strukturelle Kardiologie"
#   ./run_digest.sh dermatology check


# Lauf startest, kostenlos, Anbieter gemini
python3 clarity_digest.py --check-params --provider gemini


# Lauf startest, kostenlos, Anbieter anthropic
python3 clarity_digest.py --check-params --provider anthropic

# mit Claude (Standard, auch ohne --provider)
python3 clarity_digest.py --provider anthropic --days 7 --pdf --docx

# mit Gemini
python3 clarity_digest.py --provider gemini --days 7 --pdf --docx

#
# Danach prüft das Programm beide Modelle mit einem kurzen Testaufruf. 
# Ist der Key oder ein Modellname falsch, bricht es dort mit einer klaren Meldung ab, bevor Kosten entstehen.
# Für Tests empfehle ich, wie bisher einen eigenen Ordner, Bspw. test_state(STATE_DIR=test_state) und --no-dedup zu verwenden. 
# So bleibt dein echter State 
#
STATE_DIR=test_state python3 clarity_digest.py --provider gemini --days 7 --no-dedup --limit 40 --pdf --docx

STATE_DIR=test_state python3 clarity_digest.py --provider gemini --days 7 --no-dedup --limit 40 --pdf --docx
